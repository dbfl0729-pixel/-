import React, { useState, useEffect, useMemo, useRef } from 'react';
import { initializeApp } from 'firebase/app';
import { getAuth, signInAnonymously, signInWithCustomToken, onAuthStateChanged } from 'firebase/auth';
import { getFirestore, collection, doc, onSnapshot, setDoc, addDoc, updateDoc, deleteDoc } from 'firebase/firestore';
import { 
  Plus, Trash2, Edit3, Settings, ShoppingCart, 
  ChevronRight, AlignJustify, Calculator, X, Save,
  CheckCircle2, AlertCircle, LayoutList, Menu
} from 'lucide-react';

// ============================================================================
// Firebase Initialization & Config
// ============================================================================
const firebaseConfig = typeof __firebase_config !== 'undefined' ? JSON.parse(__firebase_config) : {};
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const appId = typeof __app_id !== 'undefined' ? __app_id : 'clinic-consult-app';

// ============================================================================
// Helper Components
// ============================================================================

// 화폐 포맷
const formatPrice = (price) => new Intl.NumberFormat('ko-KR').format(price || 0);

// 인라인 편집 컴포넌트
const EditableField = ({ value, onSave, isEditing, type = "text", multiline = false, className = "", placeholder = "" }) => {
  const [val, setVal] = useState(value);
  
  useEffect(() => { setVal(value); }, [value]);

  const handleBlur = () => {
    let finalValue = val;
    if (type === 'number') {
      finalValue = Number(val) || 0;
    }
    if (finalValue !== value) {
      onSave(finalValue);
    }
  };

  if (!isEditing) {
    if (type === 'number') return <span className={className}>{formatPrice(value)}</span>;
    return <span className={`whitespace-pre-wrap ${className}`}>{value || placeholder}</span>;
  }

  const baseInputClass = `w-full border-b-2 border-blue-400 bg-blue-50/50 focus:outline-none focus:border-blue-600 transition-colors px-1 ${className}`;

  if (multiline) {
    return (
      <textarea
        className={`${baseInputClass} resize-none overflow-hidden`}
        value={val}
        onChange={e => {
          setVal(e.target.value);
          e.target.style.height = 'auto';
          e.target.style.height = e.target.scrollHeight + 'px';
        }}
        onBlur={handleBlur}
        placeholder={placeholder}
        rows={1}
      />
    );
  }

  return (
    <input
      type={type === 'number' ? 'number' : 'text'}
      className={baseInputClass}
      value={val}
      onChange={e => setVal(e.target.value)}
      onBlur={handleBlur}
      placeholder={placeholder}
    />
  );
};

// ============================================================================
// Main Application Component
// ============================================================================
export default function App() {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  
  // Data States
  const [globalSettings, setGlobalSettings] = useState({ clinicName: '프리미엄 클리닉 상담 시스템' });
  const [categories, setCategories] = useState([]);
  const [items, setItems] = useState([]);
  
  // UI States
  const [activeCategoryId, setActiveCategoryId] = useState(null);
  const [isEditMode, setIsEditMode] = useState(false);
  const [isCartOpen, setIsCartOpen] = useState(true);
  const [isSidebarOpen, setIsSidebarOpen] = useState(true);
  
  // Cart & Calculator States
  const [cart, setCart] = useState([]);
  const [discountMode, setDiscountMode] = useState('none'); // 'none', 'preset', 'customRate', 'customPrice'
  const [discountRate, setDiscountRate] = useState(0); // 0~100 %
  const [customFinalPrice, setCustomFinalPrice] = useState(''); // 직접 입력한 최종가

  // 1. Firebase Auth Initialization (Rule 3)
  useEffect(() => {
    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (error) {
        console.error('Auth Error:', error);
      }
    };
    initAuth();

    const unsubscribe = onAuthStateChanged(auth, (currentUser) => {
      setUser(currentUser);
      if (currentUser) setLoading(false);
    });
    return () => unsubscribe();
  }, []);

  // 2. Firestore Data Fetching (Rule 1 & 2)
  useEffect(() => {
    if (!user) return;

    const baseRef = (colName) => collection(db, 'artifacts', appId, 'users', user.uid, colName);

    // Settings Listener
    const unsubSettings = onSnapshot(doc(db, 'artifacts', appId, 'users', user.uid, 'settings', 'global'), 
      (docSnap) => {
        if (docSnap.exists()) {
          setGlobalSettings(docSnap.data());
        } else {
          // Initialize Settings
          setDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'settings', 'global'), { clinicName: '청담 프리미엄 클리닉' });
        }
      },
      (err) => console.error("Settings listener error:", err)
    );

    // Categories Listener
    const unsubCategories = onSnapshot(baseRef('categories'), 
      (snapshot) => {
        const cats = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
        cats.sort((a, b) => (a.order || 0) - (b.order || 0));
        setCategories(cats);
        
        // Auto-select first category if none selected
        if (cats.length > 0 && !activeCategoryId) {
          setActiveCategoryId(cats[0].id);
        }
      },
      (err) => console.error("Categories listener error:", err)
    );

    // Items Listener
    const unsubItems = onSnapshot(baseRef('items'), 
      (snapshot) => {
        const itms = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
        itms.sort((a, b) => (a.order || 0) - (b.order || 0));
        setItems(itms);
      },
      (err) => console.error("Items listener error:", err)
    );

    return () => {
      unsubSettings();
      unsubCategories();
      unsubItems();
    };
  }, [user]);

  // ============================================================================
  // Data Handlers (Firestore Writes)
  // ============================================================================
  const updateSettings = async (newName) => {
    if (!user) return;
    await setDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'settings', 'global'), { clinicName: newName }, { merge: true });
  };

  const addCategory = async () => {
    if (!user) return;
    const newOrder = categories.length > 0 ? Math.max(...categories.map(c => c.order || 0)) + 1 : 0;
    await addDoc(collection(db, 'artifacts', appId, 'users', user.uid, 'categories'), {
      name: '새 카테고리',
      isLaserTable: false,
      order: newOrder
    });
  };

  const updateCategory = async (id, data) => {
    if (!user) return;
    await updateDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'categories', id), data);
  };

  const deleteCategory = async (id) => {
    if (!user || !window.confirm('카테고리와 포함된 모든 시술이 삭제됩니다. 계속하시겠습니까?')) return;
    // Remove items in category
    items.filter(i => i.categoryId === id).forEach(async (item) => {
      await deleteDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'items', item.id));
    });
    // Remove category
    await deleteDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'categories', id));
    if (activeCategoryId === id) setActiveCategoryId(categories[0]?.id || null);
  };

  const addItem = async (categoryId) => {
    if (!user) return;
    const catItems = items.filter(i => i.categoryId === categoryId);
    const newOrder = catItems.length > 0 ? Math.max(...catItems.map(c => c.order || 0)) + 1 : 0;
    await addDoc(collection(db, 'artifacts', appId, 'users', user.uid, 'items'), {
      categoryId,
      name: '새 시술 항목',
      price: 100000,
      effect: '시술의 주요 효과를 입력하세요.',
      details: '시술에 대한 상세 설명 및 추천 대상을 입력하세요.',
      order: newOrder
    });
  };

  const updateItem = async (id, data) => {
    if (!user) return;
    await updateDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'items', id), data);
  };

  const deleteItem = async (id) => {
    if (!user || !window.confirm('이 항목을 삭제하시겠습니까?')) return;
    await deleteDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'items', id));
  };

  // ============================================================================
  // Cart & Calculator Logic
  // ============================================================================
  const addToCart = (item) => {
    setCart([...cart, { ...item, cartId: Date.now() + Math.random() }]);
  };

  const removeFromCart = (cartId) => {
    setCart(cart.filter(item => item.cartId !== cartId));
  };

  // 계산 로직 (총합 -> 할인 -> 최종가 및 역산)
  const totalSum = cart.reduce((acc, item) => acc + (Number(item.price) || 0), 0);
  
  let finalPrice = totalSum;
  let actualDiscountAmount = 0;
  let actualDiscountRate = 0;

  if (totalSum > 0) {
    if (discountMode === 'preset' || discountMode === 'customRate') {
      actualDiscountAmount = totalSum * (discountRate / 100);
      finalPrice = totalSum - actualDiscountAmount;
      actualDiscountRate = discountRate;
    } else if (discountMode === 'customPrice') {
      finalPrice = Number(customFinalPrice) || 0;
      actualDiscountAmount = totalSum - finalPrice;
      actualDiscountRate = (actualDiscountAmount / totalSum) * 100;
    }
  }

  // ============================================================================
  // Render Components
  // ============================================================================
  if (loading) {
    return (
      <div className="flex h-screen w-full items-center justify-center bg-gray-50">
        <div className="flex flex-col items-center space-y-4">
          <div className="h-10 w-10 animate-spin rounded-full border-4 border-gray-200 border-t-slate-800"></div>
          <p className="text-gray-500 font-medium">상담 시스템을 불러오는 중입니다...</p>
        </div>
      </div>
    );
  }

  const activeCategory = categories.find(c => c.id === activeCategoryId);
  const activeItems = items.filter(i => i.categoryId === activeCategoryId);

  return (
    <div className="flex h-screen bg-[#F8FAFC] text-slate-800 font-sans overflow-hidden">
      
      {/* 1. Left Sidebar Navigation */}
      <aside className={`w-72 flex-shrink-0 bg-white border-r border-gray-200 flex flex-col shadow-sm z-20 transition-all duration-300 ease-in-out ${isSidebarOpen ? 'ml-0' : '-ml-72'}`}>
        <div className="p-6 border-b border-gray-100 bg-white">
          <EditableField
            isEditing={isEditMode}
            value={globalSettings.clinicName}
            onSave={updateSettings}
            className="text-xl font-bold tracking-tight text-slate-900"
          />
          <p className="text-sm text-gray-400 mt-1 font-medium tracking-wide">Premium Consultation</p>
        </div>

        <div className="flex-1 overflow-y-auto p-4 space-y-2">
          {categories.map((cat) => (
            <div 
              key={cat.id}
              className={`group flex items-center justify-between p-3 rounded-xl transition-all duration-200 cursor-pointer ${
                activeCategoryId === cat.id 
                  ? 'bg-slate-800 text-white shadow-md' 
                  : 'hover:bg-slate-100 text-slate-600'
              }`}
              onClick={() => setActiveCategoryId(cat.id)}
            >
              <div className="flex items-center space-x-3 flex-1 overflow-hidden">
                {cat.isLaserTable ? <AlignJustify size={18} className={activeCategoryId === cat.id ? 'text-blue-300' : 'text-slate-400'}/> : <LayoutList size={18} className={activeCategoryId === cat.id ? 'text-blue-300' : 'text-slate-400'}/>}
                <EditableField
                  isEditing={isEditMode}
                  value={cat.name}
                  onSave={(val) => updateCategory(cat.id, { name: val })}
                  className={`font-semibold truncate ${activeCategoryId === cat.id ? 'text-white' : ''}`}
                />
              </div>
              
              {isEditMode && (
                <button 
                  onClick={(e) => { e.stopPropagation(); deleteCategory(cat.id); }}
                  className="text-red-400 hover:text-red-500 opacity-0 group-hover:opacity-100 transition-opacity p-1"
                >
                  <Trash2 size={16} />
                </button>
              )}
            </div>
          ))}

          {isEditMode && (
            <button 
              onClick={addCategory}
              className="w-full flex items-center justify-center space-x-2 p-3 mt-4 border-2 border-dashed border-gray-300 rounded-xl text-gray-500 hover:bg-gray-50 hover:border-gray-400 transition-colors"
            >
              <Plus size={18} />
              <span className="font-medium">카테고리 추가</span>
            </button>
          )}
        </div>

        {/* Edit Mode Toggle Footer */}
        <div className="p-4 border-t border-gray-200 bg-gray-50">
          <button
            onClick={() => setIsEditMode(!isEditMode)}
            className={`w-full flex items-center justify-center space-x-2 py-3 rounded-xl font-semibold transition-colors ${
              isEditMode 
                ? 'bg-blue-100 text-blue-700 hover:bg-blue-200' 
                : 'bg-white border border-gray-300 text-gray-600 hover:bg-gray-100 shadow-sm'
            }`}
          >
            {isEditMode ? <Save size={18} /> : <Settings size={18} />}
            <span>{isEditMode ? '편집 완료 (자동 저장됨)' : '시스템 편집 모드'}</span>
          </button>
          <div className="text-xs text-center mt-3 text-gray-400 flex items-center justify-center">
            <CheckCircle2 size={12} className="mr-1" /> 실시간 클라우드 동기화 중
          </div>
        </div>
      </aside>

      {/* 2. Main Content Area */}
      <main className="flex-1 flex flex-col overflow-hidden bg-[#F8FAFC]">
        {activeCategory ? (
          <>
            {/* Header */}
            <header className="px-8 py-6 bg-white border-b border-gray-200 flex items-center justify-between shadow-sm z-0">
              <div className="flex items-center space-x-4">
                <button 
                  onClick={() => setIsSidebarOpen(!isSidebarOpen)}
                  className="p-2 -ml-2 text-slate-400 hover:text-slate-800 hover:bg-slate-100 rounded-xl transition-colors"
                  title="메뉴 열기/닫기"
                >
                  <Menu size={24} />
                </button>
                <h2 className="text-2xl font-bold text-slate-800 tracking-tight flex items-center">
                  <EditableField
                    isEditing={isEditMode}
                    value={activeCategory.name}
                    onSave={(val) => updateCategory(activeCategory.id, { name: val })}
                  />
                </h2>
                {isEditMode && (
                  <div className="flex items-center bg-gray-100 rounded-lg p-1 space-x-2 text-sm">
                    <label className="flex items-center space-x-2 cursor-pointer px-2 py-1 hover:bg-gray-200 rounded">
                      <input 
                        type="checkbox" 
                        checked={activeCategory.isLaserTable || false}
                        onChange={(e) => updateCategory(activeCategory.id, { isLaserTable: e.target.checked })}
                        className="rounded text-blue-600 focus:ring-blue-500"
                      />
                      <span className="text-gray-600 font-medium">단품 레이저 표 형식 사용</span>
                    </label>
                  </div>
                )}
              </div>
              
              <button 
                onClick={() => setIsCartOpen(!isCartOpen)}
                className="flex items-center space-x-2 bg-slate-800 text-white px-5 py-2.5 rounded-full hover:bg-slate-700 transition-colors shadow-md"
              >
                <ShoppingCart size={18} />
                <span className="font-semibold">패키지 구성</span>
                {cart.length > 0 && (
                  <span className="bg-blue-500 text-white text-xs px-2 py-0.5 rounded-full ml-2">
                    {cart.length}
                  </span>
                )}
              </button>
            </header>

            {/* Content Body */}
            <div className="flex-1 overflow-y-auto p-8">
              
              {/* Type A: 일반 레이아웃 (Effect & Details 확장형) */}
              {!activeCategory.isLaserTable && (
                <div className="grid grid-cols-1 xl:grid-cols-2 gap-6">
                  {activeItems.map((item) => (
                    <div key={item.id} className="bg-white rounded-2xl shadow-sm border border-gray-100 overflow-hidden hover:shadow-md transition-shadow flex flex-col">
                      <div className="p-6 border-b border-gray-50 flex justify-between items-start bg-gradient-to-r from-white to-slate-50">
                        <div className="flex-1">
                          <h3 className="text-xl font-bold text-slate-800 mb-1">
                            <EditableField isEditing={isEditMode} value={item.name} onSave={(val) => updateItem(item.id, { name: val })} />
                          </h3>
                        </div>
                        <div className="text-right ml-4">
                          <div className="text-xl font-bold text-slate-800">
                            ₩ <EditableField isEditing={isEditMode} type="number" value={item.price} onSave={(val) => updateItem(item.id, { price: val })} className="text-right" />
                          </div>
                        </div>
                      </div>
                      
                      {/* Effect & Details 확장 영역 */}
                      <div className="p-6 flex-1 flex flex-col space-y-4 bg-white">
                        <div>
                          <span className="text-xs font-bold text-blue-500 uppercase tracking-wider mb-2 block">Key Effect</span>
                          <p className="text-slate-600 font-medium text-sm leading-relaxed bg-blue-50/50 p-3 rounded-xl border border-blue-100/50">
                            <EditableField isEditing={isEditMode} multiline value={item.effect} onSave={(val) => updateItem(item.id, { effect: val })} />
                          </p>
                        </div>
                        <div className="flex-1">
                          <span className="text-xs font-bold text-gray-400 uppercase tracking-wider mb-2 block">Details</span>
                          <p className="text-gray-500 text-sm leading-relaxed">
                            <EditableField isEditing={isEditMode} multiline value={item.details} onSave={(val) => updateItem(item.id, { details: val })} />
                          </p>
                        </div>
                      </div>

                      <div className="p-4 bg-gray-50 border-t border-gray-100 flex justify-between items-center">
                        {isEditMode ? (
                          <button onClick={() => deleteItem(item.id)} className="text-red-500 hover:text-red-600 flex items-center text-sm font-medium px-3 py-1.5 rounded-lg hover:bg-red-50">
                            <Trash2 size={16} className="mr-1"/> 삭제
                          </button>
                        ) : (
                          <div /> // Spacer
                        )}
                        <button 
                          onClick={() => addToCart(item)}
                          className="bg-slate-800 text-white px-6 py-2 rounded-xl text-sm font-bold hover:bg-slate-700 transition-colors shadow-sm ml-auto"
                        >
                          패키지에 담기
                        </button>
                      </div>
                    </div>
                  ))}
                </div>
              )}

              {/* Type B: 단품 레이저 전용 테이블 레이아웃 */}
              {activeCategory.isLaserTable && (
                <div className="bg-white rounded-2xl shadow-sm border border-gray-200 overflow-hidden">
                  <table className="w-full text-left border-collapse">
                    <thead>
                      <tr className="bg-slate-800 text-white text-sm tracking-wider">
                        <th className="p-4 font-semibold w-1/4">시술명</th>
                        <th className="p-4 font-semibold w-1/4">기대 효과</th>
                        <th className="p-4 font-semibold w-1/3">상세 설명</th>
                        <th className="p-4 font-semibold w-48 text-right">금액 (VAT 별도)</th>
                        <th className="p-4 w-24 text-center">선택</th>
                      </tr>
                    </thead>
                    <tbody className="divide-y divide-gray-100 text-sm">
                      {activeItems.map((item) => (
                        <tr key={item.id} className="hover:bg-slate-50 transition-colors group">
                          <td className="p-4 align-top">
                            <div className="font-bold text-slate-800 text-base">
                              <EditableField isEditing={isEditMode} value={item.name} onSave={(val) => updateItem(item.id, { name: val })} />
                            </div>
                            {isEditMode && (
                              <button onClick={() => deleteItem(item.id)} className="text-red-400 hover:text-red-600 text-xs mt-2 flex items-center">
                                <Trash2 size={12} className="mr-1"/> 항목 삭제
                              </button>
                            )}
                          </td>
                          <td className="p-4 align-top text-slate-600 font-medium">
                            <EditableField isEditing={isEditMode} multiline value={item.effect} onSave={(val) => updateItem(item.id, { effect: val })} />
                          </td>
                          <td className="p-4 align-top text-gray-500 leading-relaxed">
                            <EditableField isEditing={isEditMode} multiline value={item.details} onSave={(val) => updateItem(item.id, { details: val })} />
                          </td>
                          <td className="p-4 align-top text-right">
                            <div className="font-bold text-lg text-slate-800">
                              ₩ <EditableField isEditing={isEditMode} type="number" value={item.price} onSave={(val) => updateItem(item.id, { price: val })} className="text-right w-24" />
                            </div>
                          </td>
                          <td className="p-4 align-middle text-center">
                            <button 
                              onClick={() => addToCart(item)}
                              className="bg-white border border-slate-300 text-slate-700 p-2 rounded-lg hover:bg-slate-800 hover:text-white hover:border-slate-800 transition-all"
                              title="패키지에 추가"
                            >
                              <Plus size={18} />
                            </button>
                          </td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                </div>
              )}

              {/* Add Item Button */}
              {isEditMode && (
                <div className="mt-8 flex justify-center">
                  <button 
                    onClick={() => addItem(activeCategory.id)}
                    className="flex items-center space-x-2 bg-white border-2 border-dashed border-gray-300 text-gray-600 px-8 py-4 rounded-2xl hover:bg-gray-50 hover:border-gray-400 transition-all shadow-sm font-semibold"
                  >
                    <Plus size={20} />
                    <span>새 시술 항목 추가</span>
                  </button>
                </div>
              )}
            </div>
          </>
        ) : (
          <div className="flex-1 flex items-center justify-center text-gray-400">
            {isEditMode ? "카테고리를 추가하여 편집을 시작하세요." : "좌측 메뉴에서 카테고리를 선택해주세요."}
          </div>
        )}
      </main>

      {/* 3. Right Sidebar: Package Calculator & Cart */}
      {isCartOpen && (
        <aside className="w-[400px] bg-white border-l border-gray-200 flex flex-col shadow-xl z-20">
          <div className="p-5 border-b border-gray-200 bg-slate-50 flex justify-between items-center">
            <h2 className="text-lg font-bold text-slate-800 flex items-center">
              <Calculator size={20} className="mr-2 text-slate-600" />
              패키지 구성 및 견적
            </h2>
            <button onClick={() => setIsCartOpen(false)} className="text-gray-400 hover:text-gray-600 p-1 rounded-md hover:bg-gray-200">
              <X size={20} />
            </button>
          </div>

          {/* Cart Items List */}
          <div className="flex-1 overflow-y-auto p-5 bg-white space-y-3">
            {cart.length === 0 ? (
              <div className="h-full flex flex-col items-center justify-center text-gray-400 space-y-3">
                <ShoppingCart size={40} className="opacity-20" />
                <p className="text-sm">선택된 시술이 없습니다.</p>
              </div>
            ) : (
              cart.map((item, index) => (
                <div key={item.cartId} className="flex justify-between items-center p-3 border border-gray-100 rounded-xl bg-gray-50/50 hover:bg-white hover:shadow-sm transition-all group">
                  <div className="flex-1 pr-3">
                    <p className="text-xs text-gray-400 mb-0.5">{categories.find(c => c.id === item.categoryId)?.name}</p>
                    <p className="font-bold text-slate-800 text-sm">{item.name}</p>
                  </div>
                  <div className="text-right flex items-center space-x-3">
                    <span className="font-bold text-slate-800">₩ {formatPrice(item.price)}</span>
                    <button 
                      onClick={() => removeFromCart(item.cartId)}
                      className="text-gray-300 hover:text-red-500 transition-colors p-1"
                    >
                      <X size={16} />
                    </button>
                  </div>
                </div>
              ))
            )}
          </div>

          {/* Calculator Section (Strict Layout Rules) */}
          <div className="border-t border-gray-200 bg-white p-6 shadow-[0_-10px_30px_-15px_rgba(0,0,0,0.1)]">
            
            {/* 1. 총합 */}
            <div className="flex justify-between items-end mb-6">
              <span className="text-sm font-bold text-gray-500">총 시술 금액</span>
              <span className="text-xl font-bold text-slate-400 text-right">
                ₩ {formatPrice(totalSum)}
              </span>
            </div>

            {/* 2. 할인 설정 패널 */}
            <div className="bg-slate-50 rounded-xl p-4 mb-6 border border-slate-100">
              <span className="text-xs font-bold text-slate-500 block mb-3">할인율 적용</span>
              
              <div className="grid grid-cols-4 gap-2 mb-3">
                {[10, 15, 20].map((rate) => (
                  <button
                    key={rate}
                    onClick={() => { setDiscountMode('preset'); setDiscountRate(rate); }}
                    className={`py-2 rounded-lg text-sm font-bold transition-colors ${
                      discountMode === 'preset' && discountRate === rate 
                        ? 'bg-slate-800 text-white' 
                        : 'bg-white border border-gray-200 text-slate-600 hover:bg-gray-100'
                    }`}
                  >
                    {rate}%
                  </button>
                ))}
                <button
                  onClick={() => { setDiscountMode('customRate'); }}
                  className={`py-2 rounded-lg text-sm font-bold transition-colors ${
                    discountMode === 'customRate'
                      ? 'bg-slate-800 text-white' 
                      : 'bg-white border border-gray-200 text-slate-600 hover:bg-gray-100'
                  }`}
                >
                  직접(%)
                </button>
              </div>

              {/* Custom Input Fields based on mode */}
              {discountMode === 'customRate' && (
                <div className="flex items-center space-x-2 mt-2">
                  <input
                    type="number"
                    value={discountRate}
                    onChange={(e) => setDiscountRate(Number(e.target.value))}
                    className="flex-1 p-2 text-right border border-gray-300 rounded-lg focus:outline-none focus:border-slate-800 font-bold"
                    placeholder="할인율 입력"
                  />
                  <span className="text-slate-600 font-bold">%</span>
                </div>
              )}

              <div className="mt-3 pt-3 border-t border-gray-200">
                <button
                  onClick={() => { setDiscountMode('customPrice'); setCustomFinalPrice(finalPrice); }}
                  className={`w-full py-2 rounded-lg text-sm font-bold transition-colors flex justify-center items-center space-x-1 ${
                    discountMode === 'customPrice'
                      ? 'bg-slate-200 text-slate-800 border border-slate-300' 
                      : 'bg-transparent text-gray-500 hover:bg-gray-100 underline decoration-gray-300 underline-offset-4'
                  }`}
                >
                  <span>최종 확정가 직접 입력 (할인율 역산)</span>
                </button>
                
                {discountMode === 'customPrice' && (
                  <div className="flex items-center mt-2 relative">
                    <span className="absolute left-3 text-slate-400 font-bold">₩</span>
                    <input
                      type="number"
                      value={customFinalPrice}
                      onChange={(e) => setCustomFinalPrice(e.target.value)}
                      className="w-full p-2 pl-8 text-right border-2 border-slate-400 rounded-lg focus:outline-none focus:border-slate-800 font-bold text-lg"
                      placeholder="최종 금액 입력"
                    />
                  </div>
                )}
              </div>
            </div>

            {/* 3. 할인 결과 Summary (명확한 우측 정렬) */}
            <div className="space-y-3 mb-6">
              <div className="flex justify-between items-center text-sm">
                <span className="text-slate-500">할인 금액</span>
                <span className="font-semibold text-red-500 text-right">
                  - ₩ {formatPrice(actualDiscountAmount)}
                </span>
              </div>
              <div className="flex justify-between items-center text-sm">
                <span className="text-slate-500">실제 적용 할인율</span>
                <span className="font-semibold text-slate-700 text-right bg-slate-100 px-2 py-0.5 rounded">
                  {totalSum > 0 ? actualDiscountRate.toFixed(1) : 0}%
                </span>
              </div>
            </div>

            {/* 4. 최종가 (유일한 강조 색상 사용) */}
            <div className="pt-4 border-t-2 border-slate-800 flex justify-between items-end">
              <span className="text-lg font-bold text-slate-800">최종 제안가</span>
              <span className="text-3xl font-extrabold text-blue-600 text-right tracking-tight">
                ₩ {formatPrice(finalPrice)}
              </span>
            </div>

            <button 
              className="w-full mt-6 bg-blue-600 hover:bg-blue-700 text-white font-bold py-4 rounded-xl shadow-lg shadow-blue-600/20 transition-all text-lg"
              onClick={() => alert(`최종가 ${formatPrice(finalPrice)}원으로 상담이 저장/진행됩니다.`)}
            >
              상담 내역 확정
            </button>
          </div>
        </aside>
      )}

    </div>
  );
}
